from pathlib import Path
import csv,json,os,subprocess as sp,time,statistics
root=Path.cwd()
work=root/'.context/bench-lib'
evidence=root/'docs/benchmarks/reports/2026-11-25-tamil-ios-drivers/evidence'
device='418CEC10-C73F-4444-AD12-07AD9CF58008'
env={**os.environ,'WF_USAGE':'off','WF_IDB_PYTHON':str(Path.home()/'.local/pipx/venvs/fb-idb/bin/python')}
def sample(pid,kind):
    rows=[]
    for line in sp.check_output(['ps','-axo','pid=,ppid=,rss=,command='],text=True).splitlines():
        fields=line.strip().split(None,3)
        if len(fields)==4: rows.append((*map(int,fields[:3]),fields[3]))
    tree={pid} if pid else set()
    while True:
        linked={p for p,parent,_,_ in rows if parent in tree}
        if linked<=tree: break
        tree.update(linked)
    tool=app=0
    for p,parent,rss,cmd in rows:
        if device in cmd and '/mlatamagui.app/mlatamagui' in cmd: app+=rss
        elif p in tree or (device in cmd and (('idb_companion' in cmd.split(' ')[0] and kind=='wf') or ('maestro' in cmd.lower().split(' ')[0] and kind=='maestro'))): tool+=rss
    return tool/1024,app/1024
results=[]
def measure(pair,kind):
    out=work/'series'/f'{pair}-{kind}';out.mkdir()
    sp.run(['idb','terminate','com.nuesslerm.mla-tamagui','--udid',device],check=True)
    with (out/'setup.log').open('w') as log:
        sp.run([str(root/'clis/target/debug/wf'),'ios','test',str(work/'preflight.lua'),'--device',device,'--test-output-dir',str(out/'preflight'),'--no-record'],env=env,stdout=log,stderr=sp.STDOUT,check=True)
    time.sleep(2)
    cmd=([str(root/'clis/target/debug/wf'),'ios','test',str(work/'test.lua'),'--device',device,'--env','out='+str(out),'--test-output-dir',str(out)] if kind=='wf' else ['maestro','--device',device,'test',str(work/'test.yaml'),'--test-output-dir',str(out),'--format','JUNIT','--output',str(out/'report.xml')])
    tool0,app0=sample(None,kind)
    with (out/'log').open('w') as log:
        wall=time.time(); start=time.monotonic(); proc=sp.Popen(cmd,env=env,stdout=log,stderr=sp.STDOUT)
        data=[]
        while proc.poll() is None:
            data.append([round(time.monotonic()-start,3),*sample(proc.pid,kind)])
            if time.monotonic()-start>180:proc.terminate();break
            time.sleep(.16)
        code=proc.wait(); total=time.monotonic()-start
    with (evidence/f'{pair}-{kind}.csv').open('w') as f:
        w=csv.writer(f);w.writerow(['elapsed_s','driver_tree_rss_mib','app_rss_mib']);w.writerows(data)
    after,_=sample(None,kind)
    result=dict(pair=pair,driver=kind,exit=code,total=total,baseline=tool0,appBaseline=app0,peak=max(x[1] for x in data)-tool0,appPeak=max(x[2] for x in data),residual=after-tool0)
    if code==0:
        def stamp(mark): return next(out.rglob(mark+'.png')).stat().st_mtime
        result.update(startup=stamp('start')-wall,journey=stamp('finish')-stamp('start'))
    results.append(result)
    (evidence/'runs.json').write_text(json.dumps(results,indent=2)+'\n')
    print(result,flush=True)
    time.sleep(5)
for pair in range(1,6):
    for kind in (['wf','maestro'] if pair%2 else ['maestro','wf']):
        if results and results[-1]["driver"]=="maestro" and kind=="maestro":time.sleep(65)
        measure(pair,kind)
